# backend/controllers/categories_controller.py
from flask import Blueprint, request, jsonify
from backend.services.categories_service import CategoriesService
from backend.repositories.categories_repository import CategoriesRepository
from backend.db_session import get_db

categories_bp = Blueprint("categories", __name__)

def _service():
    repo = CategoriesRepository(get_db())
    return CategoriesService(repo)

@categories_bp.post("/")
def create_category():
    data = request.get_json() or {}
    created = _service().create_category(data)
    return jsonify(created), 201

@categories_bp.get("/<int:category_id>")
def get_category(category_id: int):
    found = _service().get_category_by_id(category_id)
    if not found:
        return jsonify({"error": "Category not found"}), 404
    return jsonify(found), 200

@categories_bp.get("/")
def list_categories():
    items = _service().list_categories()
    return jsonify(items), 200

@categories_bp.put("/<int:category_id>")
def update_category(category_id: int):
    data = request.get_json() or {}
    updated = _service().update_category(category_id, data)
    if not updated:
        return jsonify({"error": "Category not found"}), 404
    return jsonify(updated), 200

@categories_bp.delete("/<int:category_id>")
def delete_category(category_id: int):
    ok = _service().delete_category(category_id)
    if not ok:
        return jsonify({"error": "Category not found"}), 404
    return jsonify({"message": "Category deleted"}), 200

@categories_bp.get("/search")
def search_categories():
    # 예시: /api/categories/search?name=Health&partial=true
    name = request.args.get("name")
    partial = (request.args.get("partial", "true").lower() == "true")
    if not name:
        return jsonify({"error": "Provide 'name'"}), 400
    results = _service().search_categories_by_name(name=name, partial=partial)
    return jsonify(results), 200



# backend/services/categories_service.py
from typing import Dict, Any, Optional, List
from backend.schemas.categories import Category  # Pydantic 스키마

class CategoriesService:
    def __init__(self, repository):
        self.repository = repository

    def create_category(self, data: Dict[str, Any]) -> Dict[str, Any]:
        cat = Category(**data)  # 검증
        # 중복 이름 방지 등 비즈 규칙이 있으면 여기서
        existing = self.repository.get_category_by_name(cat.name)
        if existing:
            raise ValueError("Category name already exists.")
        created = self.repository.create_category(name=cat.name, description=cat.description)
        return self.repository.get_category_by_id(created["id"])

    def get_category_by_id(self, category_id: int) -> Optional[Dict[str, Any]]:
        return self.repository.get_category_by_id(category_id)

    def list_categories(self) -> List[Dict[str, Any]]:
        return self.repository.list_categories()

    def update_category(self, category_id: int, data: Dict[str, Any]) -> Optional[Dict[str, Any]]:
        current = self.repository.get_category_by_id(category_id)
        if not current:
            return None
        # 이름 변경 시 중복 체크
        if "name" in data and data["name"]:
            other = self.repository.get_category_by_name(data["name"])
            if other and other.get("id") != category_id:
                raise ValueError("Category name already exists.")
        merged = {**current, **data}
        Category(**merged)  # 재검증
        self.repository.update_category(category_id, **data)
        return self.repository.get_category_by_id(category_id)

    def delete_category(self, category_id: int) -> bool:
        try:
            self.repository.delete_category(category_id)
            return True
        except ValueError:
            return False

    def search_categories_by_name(self, name: str, partial: bool = True) -> List[Dict[str, Any]]:
        return self.repository.search_categories_by_name(name=name, partial=partial)



# backend/repositories/categories_repository.py
class CategoriesRepository:
    def __init__(self, conn):
        self.conn = conn

    def create_category(self, name: str, description: str = ""):
        cur = self.conn.cursor()
        cur.execute(
            "INSERT INTO categories (name, description) VALUES (?, ?)",
            (name, description),
        )
        self.conn.commit()
        return {"id": cur.lastrowid, "name": name, "description": description}

    def get_category_by_id(self, category_id: int):
        cur = self.conn.cursor()
        cur.execute("SELECT * FROM categories WHERE id = ? ORDER BY id ASC", (category_id,))
        row = cur.fetchone()
        return dict(row) if row else None

    def get_category_by_name(self, name: str):
        cur = self.conn.cursor()
        cur.execute("SELECT * FROM categories WHERE name = ? COLLATE NOCASE", (name,))
        row = cur.fetchone()
        return dict(row) if row else None

    def list_categories(self):
        cur = self.conn.cursor()
        cur.execute("SELECT * FROM categories ORDER BY id ASC")
        rows = cur.fetchall()
        return [dict(r) for r in rows]

    def update_category(self, category_id: int, **updates):
        allowed = {"name", "description"}
        keys = [k for k in updates.keys() if k in allowed]
        if not keys:
            raise ValueError("No valid fields to update")
        values = [updates[k] for k in keys]
        set_clause = ", ".join(f"{k}=?" for k in keys)
        values.append(category_id)
        cur = self.conn.cursor()
        cur.execute(f"UPDATE categories SET {set_clause} WHERE id = ?", values)
        self.conn.commit()
        return {"updated_id": category_id}

    def delete_category(self, category_id: int):
        cur = self.conn.cursor()
        cur.execute("DELETE FROM categories WHERE id = ?", (category_id,))
        self.conn.commit()
        if cur.rowcount <= 0:
            raise ValueError("Category not found")
        return {"deleted_id": category_id}

    def search_categories_by_name(self, name: str, partial: bool = True):
        cur = self.conn.cursor()
        if partial:
            cur.execute("SELECT * FROM categories WHERE name LIKE ? COLLATE NOCASE", (f"%{name}%",))
        else:
            cur.execute("SELECT * FROM categories WHERE name = ? COLLATE NOCASE", (name,))
        rows = cur.fetchall()
        return [dict(r) for r in rows]
